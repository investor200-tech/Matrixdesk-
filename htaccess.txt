# MatrixDesk website on Whogohost. Upload this file to public_html and rename it to  .htaccess
Options -Indexes
DirectoryIndex index.html

<IfModule mod_rewrite.c>
RewriteEngine On

# Always use the secure address
RewriteCond %{HTTPS} !=on
RewriteCond %{HTTP:X-Forwarded-Proto} !https
RewriteRule ^ https://%{HTTP_HOST}%{REQUEST_URI} [L,R=301]

# Addresses that your phone alerts, Flutterwave, Instagram and Facebook send people back to
RewriteRule ^dashboard/?$ /dashboard.html [L,QSA]
RewriteRule ^billing/verify/?$ /index.html [L,QSA]
RewriteRule ^connect/(instagram|facebook)/?$ /dashboard.html [L,QSA]
RewriteRule ^prepay/verify/?$ /index.html [L,QSA]
</IfModule>

# The dashboard may use the microphone (voice notes for the knowledge book) and nothing else
<IfModule mod_headers.c>
Header always set X-Content-Type-Options "nosniff"
Header always set Referrer-Policy "strict-origin-when-cross-origin"
Header always set Permissions-Policy "microphone=(self), camera=(), geolocation=()"
Header always set Strict-Transport-Security "max-age=31536000"
<FilesMatch "\.(html|js)$">
Header set Cache-Control "no-cache, must-revalidate"
</FilesMatch>
<Files "sw.js">
Header set Service-Worker-Allowed "/"
</Files>
</IfModule>

<IfModule mod_deflate.c>
AddOutputFilterByType DEFLATE text/html text/css application/javascript application/json
</IfModule>
